import hashlib,json,os,select,subprocess,time
from pathlib import Path
root=Path('/home/llocphann/Hadalis')
sha=subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip()
binary=root/'native/target/release/inir-companiond'
proc=subprocess.Popen([str(binary)],stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=subprocess.PIPE,bufsize=0)
pending=b''; sequence=0; states=[]
def read_until(predicate,seconds):
 global pending
 end=time.monotonic()+seconds
 while time.monotonic()<end:
  ready,_,_=select.select([proc.stdout],[],[],min(.25,max(0,end-time.monotonic())))
  if not ready: continue
  data=os.read(proc.stdout.fileno(),65536)
  assert data,'daemon exited early'
  pending+=data
  while b'\n' in pending:
   line,pending=pending.split(b'\n',1)
   state=json.loads(line); states.append(state)
   if predicate(state): return state
 raise AssertionError('no expected state before bounded deadline')
def send(kind,**fields):
 global sequence
 sequence+=1
 data=dict(v=1,seq=sequence,type=kind,**fields)
 proc.stdin.write((json.dumps(data)+'\n').encode()); proc.stdin.flush()
def cpu_ticks():
 stat=Path(f'/proc/{proc.pid}/stat').read_text().split(') ',1)[1].split()
 return int(stat[11])+int(stat[12])
def pss():
 for line in Path(f'/proc/{proc.pid}/smaps_rollup').read_text().splitlines():
  if line.startswith('Pss:'): return int(line.split()[1])
try:
 initial=read_until(lambda s:s.get('visibility')=='hidden',3)
 send('preferences',personality='energetic',appearance_frequency='always')
 read_until(lambda s:s.get('personality')=='energetic',3)
 send('event',event='show')
 read_until(lambda s:s.get('visibility')=='present',3)
 begin=time.monotonic()
 wander=read_until(lambda s:s.get('travel_id',0)>0,40)
 elapsed=time.monotonic()-begin
 assert 0<=wander['travel_target']<=1
 send('event',event='hide')
 hidden=read_until(lambda s:s.get('visibility')=='hidden',3)
 assert hidden['travel_id']==0
 before=cpu_ticks(); count=len(states); begin=time.monotonic()
 assert not pending
 ready,_,_=select.select([proc.stdout],[],[],2)
 assert not ready,'hidden daemon emitted autonomous state'
 after=cpu_ticks(); quiet=time.monotonic()-begin
 assert after==before,'hidden daemon accumulated CPU ticks'
 report=dict(schema_version=1,source_sha=sha,daemon_sha256=hashlib.sha256(binary.read_bytes()).hexdigest(),
  scope='one owned release daemon with stdio; not QML/GPU or desktop measurements',
  wander_observed=True,wander_after_seconds=elapsed,travel_id=wander['travel_id'],target_in_range=True,
  received_state_count=count,hidden_travel_id=hidden['travel_id'],quiet_seconds=quiet,
  hidden_cpu_tick_delta=after-before,clock_ticks_per_second=os.sysconf('SC_CLK_TCK'),sampled_pss_kib=pss(),status='PASS')
 (root/'docs/aqua-visual/locomotion-20261003/native-wander-quiet.json').write_text(json.dumps(report,indent=2)+'\n')
 print(json.dumps(report))
finally:
 proc.stdin.close()
 try: proc.wait(timeout=3)
 except subprocess.TimeoutExpired: proc.terminate(); proc.wait(timeout=3)
 assert not proc.stderr.read(), 'daemon emitted unexpected diagnostics'
